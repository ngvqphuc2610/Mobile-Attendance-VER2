-- WARNING: This schema is for context only and is not meant to be run.
-- Table order and constraints may not be valid for execution.

CREATE TABLE public.attendance (
  id bigint NOT NULL DEFAULT nextval('attendance_id_seq'::regclass),
  user_id uuid,
  method USER-DEFINED NOT NULL,
  confidence_score double precision CHECK (confidence_score >= 0::double precision AND confidence_score <= 1::double precision),
  at_time timestamp with time zone NOT NULL DEFAULT now(),
  note text,
  CONSTRAINT attendance_pkey PRIMARY KEY (id),
  CONSTRAINT attendance_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.classes (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  code text NOT NULL UNIQUE,
  name text NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  faculty_id uuid,
  cohort_id uuid,
  CONSTRAINT classes_pkey PRIMARY KEY (id),
  CONSTRAINT classes_faculty_id_fkey FOREIGN KEY (faculty_id) REFERENCES public.faculties(id),
  CONSTRAINT classes_cohort_id_fkey FOREIGN KEY (cohort_id) REFERENCES public.cohorts(id)
);
CREATE TABLE public.cohorts (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  year smallint NOT NULL UNIQUE,
  name text,
  CONSTRAINT cohorts_pkey PRIMARY KEY (id)
);
CREATE TABLE public.face_embeddings (
  user_id uuid NOT NULL,
  vectors jsonb NOT NULL,
  samples_count integer DEFAULT jsonb_array_length(vectors),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT face_embeddings_pkey PRIMARY KEY (user_id),
  CONSTRAINT face_embeddings_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.faculties (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  code text NOT NULL UNIQUE,
  name text NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT faculties_pkey PRIMARY KEY (id)
);
CREATE TABLE public.mssv_scheme_rules (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  cohort_from smallint NOT NULL,
  cohort_to smallint NOT NULL,
  track_code text NOT NULL,
  faculty_id uuid,
  program_track_id uuid NOT NULL,
  priority integer NOT NULL DEFAULT 100,
  active boolean NOT NULL DEFAULT true,
  note text,
  CONSTRAINT mssv_scheme_rules_pkey PRIMARY KEY (id),
  CONSTRAINT mssv_scheme_rules_faculty_id_fkey FOREIGN KEY (faculty_id) REFERENCES public.faculties(id),
  CONSTRAINT mssv_scheme_rules_program_track_id_fkey FOREIGN KEY (program_track_id) REFERENCES public.program_tracks(id)
);
CREATE TABLE public.mssv_serial_counters (
  cohort_year smallint NOT NULL,
  track_code text NOT NULL,
  faculty_id uuid NOT NULL,
  next_serial integer NOT NULL DEFAULT 1,
  CONSTRAINT mssv_serial_counters_pkey PRIMARY KEY (cohort_year, track_code, faculty_id)
);
CREATE TABLE public.profiles (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  code text NOT NULL UNIQUE,
  full_name text NOT NULL,
  class_id uuid,
  is_active boolean NOT NULL DEFAULT true,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  email text,
  phone text,
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_class_id_fkey FOREIGN KEY (class_id) REFERENCES public.classes(id)
);
CREATE TABLE public.program_tracks (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  code text NOT NULL UNIQUE,
  name text NOT NULL,
  description text,
  CONSTRAINT program_tracks_pkey PRIMARY KEY (id)
);
CREATE TABLE public.students (
  profile_id uuid NOT NULL,
  class_id uuid,
  mssv text UNIQUE,
  mssv_cohort smallint,
  mssv_track_code text,
  mssv_serial integer,
  CONSTRAINT students_pkey PRIMARY KEY (profile_id),
  CONSTRAINT students_profile_id_fkey FOREIGN KEY (profile_id) REFERENCES public.profiles(id),
  CONSTRAINT students_class_id_fkey FOREIGN KEY (class_id) REFERENCES public.classes(id)
);
CREATE TABLE public.teachers (
  profile_id uuid NOT NULL,
  faculty_id uuid,
  title text,
  office text,
  CONSTRAINT teachers_pkey PRIMARY KEY (profile_id),
  CONSTRAINT teachers_profile_id_fkey FOREIGN KEY (profile_id) REFERENCES public.profiles(id),
  CONSTRAINT teachers_faculty_id_fkey FOREIGN KEY (faculty_id) REFERENCES public.faculties(id)
);
CREATE TABLE public.user_roles (
  user_id uuid NOT NULL,
  role text NOT NULL CHECK (role = ANY (ARRAY['student'::text, 'teacher'::text, 'admin'::text])),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT user_roles_pkey PRIMARY KEY (user_id),
  CONSTRAINT user_roles_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);
